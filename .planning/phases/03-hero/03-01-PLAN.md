    Create src/components/home/LightRays.tsx:
    ```tsx
    "use client";

    import { motion } from "framer-motion";

    export function LightRays() {
      const rays = Array.from({ length: 12 }, (_, i) => ({
        id: i,
        angle: -60 + i * 10,
        opacity: 0.03 + Math.random() * 0.02,
        delay: i * 0.05,
      }));

      return (
        <div className="absolute inset-0 overflow-hidden pointer-events-none">
          {/* Radial gradient glow */}
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            transition={{ duration: 1.2, ease: "easeOut" }}
            className="absolute top-0 left-1/2 -translate-x-1/2 w-[800px] h-[600px]"
            style={{
              background:
                "radial-gradient(ellipse at 50% 0%, rgba(255,255,255,0.10) 0%, transparent 55%)",
            }}
          />

          {/* SVG diagonal lines */}
          <svg
            className="absolute inset-0 w-full h-full"
            preserveAspectRatio="none"
            viewBox="0 0 100 100"
          >
            {rays.map((ray) => (
              <motion.line
                key={ray.id}
                x1="50"
                y1="0"
                x2={50 + Math.tan((ray.angle * Math.PI) / 180) * 100}
                y2="100"
                stroke="white"
                strokeWidth="0.1"
                initial={{ opacity: 0 }}
                animate={{ opacity: ray.opacity }}
                transition={{ duration: 1.5, delay: ray.delay }}
              />
            ))}
          </svg>
        </div>
      );
    }
    ```
