    ```tsx
    "use client";

    import { motion } from "framer-motion";
    import Link from "next/link";

    interface Job {
      id: string;
      title: string;
      company: string;
      location: string;
      tags: string[];
      matchScore: number;
      gradient: string;
    }

    interface JobPreviewCardProps {
      job: Job;
    }

    export function JobPreviewCard({ job }: JobPreviewCardProps) {
      return (
        <motion.div
          initial={{ opacity: 0, x: 20 }}
          animate={{ opacity: 1, x: 0 }}
          exit={{ opacity: 0, x: -20 }}
          transition={{ duration: 0.4, ease: "easeInOut" }}
          className="relative h-full min-h-[400px] rounded-2xl overflow-hidden border border-white/[0.08] p-8 flex flex-col"
          style={{ background: job.gradient }}
        >
          {/* Company logo placeholder */}
          <div className="w-12 h-12 rounded-full bg-white/10 backdrop-blur-sm flex items-center justify-center mb-6">
            <span className="font-[family-name:var(--font-playfair)] text-lg text-white">
              {job.company.charAt(0)}
            </span>
          </div>

          {/* Job title + company */}
          <h3 className="font-[family-name:var(--font-playfair)] text-2xl text-white mb-2">
            {job.title}
          </h3>
          <p className="font-[family-name:var(--font-dm-sans)] text-sm text-white/70 mb-6">
            {job.company} — {job.location}
          </p>

          {/* Match score badge */}
          <div className="inline-flex items-center self-start px-3 py-1.5 bg-white/10 backdrop-blur-sm rounded-full mb-6">
            <span className="font-[family-name:var(--font-jetbrains)] text-sm text-white">
              {job.matchScore}% skill match
            </span>
          </div>

          {/* Skill pills */}
          <div className="flex flex-wrap gap-2 mb-auto">
            {job.tags.map((tag) => (
              <span
                key={tag}
                className="px-3 py-1 text-xs font-medium text-white/80 bg-white/10 rounded-full"
              >
                {tag}
              </span>
            ))}
          </div>

          {/* CTA */}
          <Link
            href="#"
            className="mt-6 inline-flex items-center text-sm font-medium text-white hover:text-white/80 transition-colors group"
          >
            Apply Now
            <span className="ml-1 group-hover:translate-x-1 transition-transform">→</span>
          </Link>
        </motion.div>
      );
    }
    ```
