    ```tsx
    "use client";

    import { useState, useEffect } from "react";

    export function LiveClock() {
      const [time, setTime] = useState<string>("");

      useEffect(() => {
        const updateTime = () => {
          const now = new Date();
          setTime(
            now.toLocaleTimeString("en-US", {
              hour: "2-digit",
              minute: "2-digit",
              second: "2-digit",
              hour12: false,
            })
          );
        };

        updateTime();
        const interval = setInterval(updateTime, 1000);

        return () => clearInterval(interval);
      }, []);

      return (
        <div className="mt-4">
          <p className="font-[family-name:var(--font-jetbrains)] text-[11px] uppercase tracking-wider text-[#555555] mb-1">
            Your Local Time
          </p>
          <p className="font-[family-name:var(--font-jetbrains)] text-[13px] text-[#555555]">
            {time || "--:--:--"}
          </p>
        </div>
      );
    }
    ```
