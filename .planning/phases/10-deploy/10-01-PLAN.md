      },

    ```

    ```

    /.next
    /out

<task type="auto">
  <name>Task 2: Build and verify</name>
  <files>package.json</files>
  <action>
    1. Run production build:
    ```bash
    npm run build
    ```

    2. Verify build output:
    ```bash
    ls dist/
    ls dist/index.html
    ```

    3. Check for common issues:
    - No 404 errors in build log
    - All routes generated (index.html exists)
    - CSS and JS files present in dist/_next/
    - No missing asset warnings

    4. Verify the build is self-contained:
    ```bash
    ls dist/_next/static/
    ```

    5. Optional: Test with a local static server:
    ```bash
    npx serve dist
    ```
  </action>
  <verify>
    <automated>ls dist/index.html && ls dist/_next/static/ | head -5</automated>
  </verify>
  <done>Build succeeds, dist/ directory contains index.html and all static assets.</done>
